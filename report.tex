\begin{frame}{Motivation: The Challenge of Data Valuation}
    \begin{alertblock}{Goal of Data Valuation}
        Quantify the contribution of each training data point to a model.
    \end{alertblock}

    \begin{exampleblock}{Limitations of Existing Methods}
        \begin{itemize}\small
            \item \textbf{Post-hoc Analysis:} They evaluate data value only \textit{after} the model is fully trained. This prevents the early detection of `bad' data.
                  \vspace{0.5em}
            \item \textbf{High Computational Cost:} They rely on expensive model retraining (like Leave-One-Out) or complex approximations (like Influence Functions), making them unsuitable for real-time scenarios.
                  \vspace{0.5em}
            \item \textbf{Erroneous Judgment:} They often misjudge difficult but informative data points (e.g., decision boundary) as having low value.
        \end{itemize}
    \end{exampleblock}

    \begin{block}{Our Solution: LiveVal}
        We propose \textbf{LiveVal}, a real-time and trajectory-based data valuation method designed to overcome these challenges.
    \end{block}
\end{frame}

% Page 4: Conclusion
\begin{frame}{Conclusion}
    \begin{block}{Summary}
        \textbf{LiveVal} is a novel, real-time data valuation method that uses \textbf{adaptive reference points} along the optimization trajectory. It provides:
        \begin{itemize}
            \item \textbf{Early detection} of harmful data.
            \item \textbf{Significant computational savings} in both time and memory.
            \item \textbf{Reliable and fair} valuation, proven by theory and extensive experiments.
        \end{itemize}
    \end{block}

    \vfill

    \begin{alertblock}{Key Contributions}
        \begin{enumerate}
            \item First real-time data valuation method with theoretical guarantees
            \item Adaptive reference points for accurate gradient-based assessment
            \item Memory-efficient implementation with dual-queue architecture
        \end{enumerate}
    \end{alertblock}
