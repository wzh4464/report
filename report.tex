\section{Overview}
%============================================================
\begin{frame}
  \frametitle{Overview}
  \textbf{Goal:} Integrate CodeReview as the 4th Agent (alongside Coder, QA, Miner)

  \vspace{0.4em}
  \begin{columns}[T]
    \column{0.48\textwidth}
    \textbf{Two Exposure Modes}
    \begin{itemize}\small
      \item TUI/CLI (Textual-based UI)
      \item REST API (FastAPI)
    \end{itemize}

    \column{0.48\textwidth}
    \textbf{Two Review Modes}
    \begin{itemize}\small
      \item \textbf{Lightweight:} Scenario detection $\to$ opinions $\to$ report
      \item \textbf{Deep:} + static analysis + fix generation
    \end{itemize}
  \end{columns}

  \vspace{0.6em}
  \centering
  \textbf{Core Three-Step Pipeline:}\\[0.3em]

    \node[box,right=0.8cm of b](c){Fix / Report};

  \end{tikzpicture}
\end{frame}

    Tree-sitter AST $\to$
    Category matching (INDEPENDENT scenes $\parallel$) $\to$
    LLM opinion generation $\to$
    structured report
  }

\section{API \& Configuration}
%============================================================
\begin{frame}
  \frametitle{REST API Endpoints}
  {\small
  \begin{tabular}{@{}lll@{}}
    \toprule
    \textbf{Endpoint} & \textbf{Type} & \textbf{Description} \\
    \midrule
    \texttt{POST /review}              & One-shot & Lightweight review \\
    \texttt{POST /deepreview}          & One-shot & Deep review + fixes \\
    \midrule
    \texttt{POST /routing/task/location} & Step 1 & Locate issues \\
    \texttt{POST /routing/task/opinion}  & Step 2 & Extract opinions \\
    \texttt{POST /routing/task/fix}      & Step 3 & Generate fixes \\
    \midrule
    \texttt{POST /routing/opinion}     & Combo & Steps 1+2 \\
    \texttt{POST /routing/fix}         & Combo & Steps 1+2+3 \\
    \bottomrule
  \end{tabular}}

  \vspace{0.4em}
  {\scriptsize
    Task state: in-memory store (no SQL), linked by \texttt{task\_id}.
    Served via \texttt{uvicorn} on port 8001.
    CORS enabled.
  }
\end{frame}

\section{Implementation}
%============================================================
\begin{frame}
  \frametitle{Implementation Phases}
  \begin{description}[\scriptsize Phase 7]\scriptsize
    \item[Phase 1] \textbf{Foundation} ---
      Directory structure, migrate CategoryManager + BaseCategory,
      modular scene directory, 10--15 core scenes
    \item[Phase 2] \textbf{Executors} ---
      RecursiveExecutor adapter, RequestHandler, Locator,
      OpinionExtractor, config system
    \item[Phase 3] \textbf{TUI} ---
      Input/Progress/Results screens, WorkflowRunner,
      home screen card integration
    \item[Phase 4] \textbf{Deep Mode} ---
      CodeAnalyzer, FixGenerator, ReportGenerator,
      deep workflow wiring
    \item[Phase 5] \textbf{REST API} ---
      FastAPI endpoints, in-memory task store, CORS
    \item[Phase 6] \textbf{Testing} ---
      Unit + integration tests, documentation
    \item[Phase 7] \textbf{Full Migration} ---
      Remaining 40+ scene classes, comprehensive validation
  \end{description}
\end{frame}

%------------------------------------------------------------
\begin{frame}
  \frametitle{Risks \& Future Directions}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \textbf{Risks / Mitigation}
    \begin{itemize}\scriptsize
      \item Tree-sitter version sync\\
        $\to$ reuse \texttt{language-pack}
      \item LLM prompt tuning cost\\
        $\to$ reuse existing templates
      \item Scene loading performance\\
        $\to$ on-demand modular loading
      \item No SQL for task state\\
        $\to$ memory + file-persist
      \item Token overflow risk\\
        $\to$ per-executor estimation
    \end{itemize}

    \column{0.5\textwidth}
    \textbf{Future Extensions}
    \begin{itemize}\scriptsize
      \item CI/CD Git Hook auto-review
      \item LSP server for IDE integration
      \item Custom user-defined scene rules
      \item Auto-apply fixes to PRs
      \item Review history \& trend analysis
    \end{itemize}
  \end{columns}
\end{frame}
