      \footnotesize
      \begin{itemize}\itemsep0em
        \item Unsupervised learning technique for pattern recognition
        \item Objective: Maximize intra-cluster similarity and inter-cluster dissimilarity
        \item Key algorithms: K-means, Hierarchical, DBSCAN, GMM
      \end{itemize}

      \footnotesize
      \begin{enumerate}\itemsep0em
        \footnotesize
        \item Global feature relevance assumption
              \begin{itemize}\itemsep0em

                \item Example: Gene expression patterns in specific conditions

\begin{frame}
  \frametitle{Co-Clustering Overview}
  \begin{columns}
    \column{0.55\textwidth}
    \begin{block}{\footnotesize Overview}
      \footnotesize
      \begin{itemize}
        \item Simultaneously clusters rows and columns of a matrix
        \item Captures relationships between data points and features
        \item Key algorithms: Spectral Co-clustering, NMF-based methods, Probabilistic models
      \end{itemize}
    \end{block}
    \begin{block}{\footnotesize Formal Definition}
      \footnotesize
      Given a matrix $X$ of size $m \times n$, find $K$ row clusters $R = \{R_1, ..., R_K\}$ and $L$ column clusters $C = \{C_1, ..., C_L\}$ such that:
      \begin{enumerate}
        \item $R_i \neq \emptyset$ for all $i = 1, ..., K$
        \item $C_j \neq \emptyset$ for all $j = 1, ..., L$
        \item $X_{ij} \in R_i \times C_j$
      \end{enumerate}
      Objective: Minimize $\sum_{i=1}^K \sum_{j=1}^L \text{similarity}(R_i, C_j)$
    \end{block}

\section{Problem Statement}

\begin{frame}
  \frametitle{Applications and Challenges in Co-clustering Large-Scale Data}
  \begin{columns}
    \column{0.5\textwidth}
    \textbf{Applications:}
    \begin{itemize}
      \item Text mining:
            \begin{itemize}
              \item Clustering documents and words
            \end{itemize}
      \item Collaborative filtering:
            \begin{itemize}
              \item User and item clustering for recommendations
            \end{itemize}
      \item Bioinformatics:
            \begin{itemize}
              \item Grouping genes and conditions in gene expression data
            \end{itemize}
    \end{itemize}

    \column{0.5\textwidth}
    \textbf{Key Challenges:}
    \begin{enumerate}
      \item \textbf{High Computational Complexity}
            \begin{itemize}
              \item Exponentially grows with data size
            \end{itemize}

      \item \textbf{Significant Communication Overhead}
            \begin{itemize}
              \item Frequent exchange of intermediate results
              \item Continuous updates in iterative optimization
            \end{itemize}

      \item \textbf{Dependency on Sparse Matrices}
            \begin{itemize}
              \item Algorithms optimized for sparse matrices
              \item Real-world data often dense
            \end{itemize}
    \end{enumerate}
  \end{columns}
\end{frame}

\section{Methodology}

\begin{frame}
  \frametitle{Large Matrix Partitioning}
  \textbf{Addresses:} High Computational Complexity

  \vspace{0.5cm}
  \textbf{Difficulties:}
  Preserving co-clusters during matrix partitioning is difficult, risking fragmentation of significant patterns.

  \vspace{0.5cm}
  \begin{itemize}
    \item Probabilistic model for optimal partitioning
    \item Theorem: Lower bound on detection probability
          \[ P \ge 1 - \exp \left\{ -2 T_p [\phi m (s^{(k)})^2 + \psi n (t^{(k)})^2] \right\} \]
    \item Balances computational efficiency and co-cluster integrity
    \item Enables parallel processing of smaller submatrices
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Hierarchical Co-cluster Merging}
  \begin{columns}
    \column{0.5\textwidth}
    \footnotesize % Made text footnotesize
    \textbf{Addresses:} Significant Communication Overhead

    \textbf{Difficulties:} Frequent intermediate result exchanges cause high communication overhead and heavy computation load on the master node.

    \textbf{Features of Hierarchical Co-Cluster Merging:}
    \begin{itemize}
      \item Iteratively merges clusters, starting with the most similar ones
      \item Adjusts similarity thresholds adaptively during merging
      \item Reduces the frequency of intermediate result exchanges
      \item Maintains consistency and minimizes redundancy across clusters
      \item Alleviates communication and computational load by streamlining the merging process
    \end{itemize}
