\section{Preprocessing the Data}
\begin{frame}
  \frametitle{Preprocessing the Data}
  \begin{enumerate}
    \item Estimate velocity and acceleration:
          \begin{itemize}
            \item Position: $\vec{r}(t) = (x(t), y(t))$
            \item Velocity: $\vec{v}(t) \approx \frac{-\vec{r}(t+2\Delta t) + 4\vec{r}(t+\Delta t) - 3\vec{r}(t)}{2\Delta t}$
            \item Acceleration: $\vec{a}(t) \approx \frac{\vec{r}(t+\Delta t) - 2\vec{r}(t) + \vec{r}(t-\Delta t)}{\Delta t^2}$
          \end{itemize}
    \item Split data into $n'$ time windows: $[t_i, t_i + \Delta T]$
    \item For each window, model cell trajectory as a random walk:
          \begin{equation*}
            \vec{r}(t) = \vec{\tilde{r}}(t) + \vec{\xi}(t)
          \end{equation*}
          where $\vec{\xi}(t)$ is a random displacement
    \item Generate features from trajectories:
          \begin{itemize}
            \item Mean squared displacement: $\text{MSD}(\tau) = \langle |\vec{r}(t+\tau) - \vec{r}(t)|^2 \rangle$
            \item Diffusion coefficient: $D = \frac{\text{MSD}(\tau)}{2d\tau}$, where $d$ is dimensionality
            \item Directional persistence: $P(\tau) = \langle \cos\theta(\tau) \rangle$
          \end{itemize}
  \end{enumerate}
\end{frame}

\section{Feature Matrix Creation and Challenge}
\begin{frame}
  \frametitle{Feature Matrix Creation and Challenge}
  \begin{columns}
    \column{0.6\textwidth}
    \begin{itemize}
      \item Feature matrix $F_{ij}$:
            \begin{itemize}
              \item Rows $i$: individual cells
              \item Columns $j$: trajectory features, \\ $p$ features per time window
            \end{itemize}
      \item Features may include:
            \begin{itemize}
              \item Mean velocity: $\bar{v} = \frac{1}{T}\sum_{t=1}^T v(t)$
              \item Acceleration variance: $\sigma_a^2 = \frac{1}{T}\sum_{t=1}^T (a(t) - \bar{a})^2$
              \item Trajectory curvature: $\kappa = \frac{|\dot{r} \times \ddot{r}|}{|\dot{r}|^3}$
            \end{itemize}
      \item Challenge: NaN values due to cell lifecycle
            \begin{itemize}
              \item Cell birth: NaN in earlier time windows
              \item Cell death: NaN in later time windows
            \end{itemize}
      \item Traditional co-clustering methods assume complete data
    \end{itemize}

\section{Proposed Solution}
\begin{frame}
  \frametitle{Proposed Solution: Adaptive Co-clustering}
  % \begin{columns}
  % \column{0.55\textwidth}
  \begin{enumerate}
    \item Matrix Decomposition:
          \begin{itemize}
            \item Split full matrix $F$ into submatrices $S_i$ without NaN
            \item $F = \{S_1, S_2, ..., S_k\}$
          \end{itemize}
    \item Submatrix Co-clustering:
          \begin{itemize}
            \item Apply traditional method to each $S_i$
            \item Result: $C_i = \text{CoClustering}(S_i)$
          \end{itemize}
    \item Merge Co-clusters:
          \begin{itemize}
            % \item Use similarity measure $\sigma(C_i, C_j)$
            % \item Merge if $\sigma(C_i, C_j) > \text{threshold}$
            \item Merge co-clusters using methods proposed in previous work
          \end{itemize}
  \end{enumerate}
  \vspace{0.5cm}
  Advantages:
  \begin{itemize}
    \item Handles incomplete data
    \item Preserves temporal structure
    \item Scalable to large datasets

\section{Feature Selection}
\begin{frame}
  \frametitle{Feature Selection from Co-Clusters}
  \begin{enumerate}
    \item Co-cluster Analysis:
          \begin{itemize}
            \item Compute probability of each expression class in co-clusters
          \end{itemize}
    \item Feature Importance Scoring:
          \begin{itemize}
            \item Retain co-clusters with sufficient representation from all classes
          \end{itemize}
    \item Selection Criteria:
          \begin{itemize}
            \item Select features unique to retained co-clusters
            \item Discard features shared across many co-clusters
          \end{itemize}
  \end{enumerate}

  \vspace{0.5cm}
  \textbf{Result:} Selected features correspond to discriminative cell behaviors

  \vspace{0.5cm}
  \textbf{Key Benefits:}
  \begin{itemize}
    \item Identifies discriminative features
    \item Reduces dimensionality
    \item Enhances interpretability
  \end{itemize}
\end{frame}
