%% Slide 1: Introduction & Problem
\begin{frame}
\frametitle{\small Introduction \& Problem Statement}
\vspace{-0.2cm}
\begin{columns}[t]
\begin{column}{0.48\textwidth}
\textbf{\scriptsize Background}
\begin{itemize}
  \item[\tiny$\bullet$] {\tiny Code vulnerabilities threaten software security}
  \item[\tiny$\bullet$] {\tiny LLMs show promise for vulnerability detection}
  \item[\tiny$\bullet$] {\tiny CWE taxonomy: hierarchical (130 types in 10 categories)}
\end{itemize}

\vspace{0.1cm}
\textbf{\scriptsize Key Challenges}
\begin{enumerate}
  \item[{\tiny 1.}] {\tiny \textbf{Heterogeneity}: Single model struggles with diverse vulnerability patterns (buffer overflow vs injection)}
  \item[{\tiny 2.}] {\tiny \textbf{Scalability}: Manual prompt engineering for 100+ CWE types impractical}
  \item[{\tiny 3.}] {\tiny \textbf{Hallucination}: Multi-agent systems amplify errors through inter-agent communication}
  \item[{\tiny 4.}] {\tiny \textbf{Efficiency}: Querying all agents prohibitively expensive}
\end{enumerate}
\end{column}

\begin{column}{0.48\textwidth}
\textbf{\scriptsize Proposed Solution: MulVul}
\begin{itemize}
  \item[\tiny$\star$] {\tiny \textbf{Router-Detector Architecture}: Coarse-to-fine strategy aligned with CWE hierarchy}
  \item[\tiny$\star$] {\tiny \textbf{Cross-Model Prompt Evolution}: Generator LLM (Claude) proposes prompts, Executor LLM (GPT-4o) evaluates}
  \item[\tiny$\star$] {\tiny \textbf{Retrieval Augmentation}: SCALE-based knowledge base grounds reasoning}
\end{itemize}

\vspace{0.15cm}
\textbf{\scriptsize Main Contributions}
\begin{itemize}
  \item[\tiny$\diamond$] {\tiny Novel multi-agent framework with specialized tool-augmented reasoning}
  \item[\tiny$\diamond$] {\tiny Automated prompt optimization mitigating self-correction bias}
  \item[\tiny$\diamond$] {\tiny SOTA: 34.79\% Macro-F1 (+41.5\% relative improvement)}
\end{itemize}
\end{column}
\end{columns}
\end{frame}

\begin{column}{0.42\textwidth}
\textbf{\scriptsize Offline Preparation}
\begin{itemize}
  \item[\tiny 1.] {\tiny \textbf{Knowledge Base}: Convert training samples to SCALE representations $\mathcal{K} = \{(T(x_i), y_i)\}$}
  \item[\tiny 2.] {\tiny \textbf{Prompt Evolution}:
    \begin{itemize}
      \item[\tiny-] Stage I: Optimize Router for Recall@k
      \item[\tiny-] Stage II: Optimize Detectors for F1
      \item[\tiny-] Generator ($\mathcal{M}_{\text{evo}}$) mutates prompts
      \item[\tiny-] Executor ($\mathcal{M}_{\text{exec}}$) evaluates fitness
    \end{itemize}}
\end{itemize}

\vspace{0.1cm}
\textbf{\scriptsize Online Detection}
\begin{itemize}
  \item[\tiny 1.] {\tiny \textbf{Router Agent}: SCALE analysis + Global retrieval $\to$ Top-k categories}
  \item[\tiny 2.] {\tiny \textbf{Detector Agents}: Contrastive retrieval (in-category + clean + hard negatives) $\to$ Fine-grained CWE types}
  \item[\tiny 3.] {\tiny \textbf{Isolation}: No inter-agent communication prevents error cascading}
\end{itemize}
\end{column}
\end{columns}

\begin{column}{0.48\textwidth}
\textbf{\scriptsize Key Findings}
\begin{itemize}
  \item[\tiny$\blacktriangleright$] {\tiny \textbf{SOTA Performance}: 34.79\% F1, outperforms best baseline by 41.5\%}
  \item[\tiny$\blacktriangleright$] {\tiny \textbf{Prompt Evolution}: +51.6\% improvement over manual prompts}
  \item[\tiny$\blacktriangleright$] {\tiny \textbf{Few-Shot Strength}: 48\% F1 on CWEs with <100 samples (2.4$\times$ vs LLM$\times$CPG)}
  \item[\tiny$\blacktriangleright$] {\tiny \textbf{Routing Trade-off}: Top-3 balances precision (27.90\%) and recall (46.19\%)}
  \item[\tiny$\blacktriangleright$] {\tiny \textbf{Component Analysis}:
    \begin{itemize}
      \item[\tiny-] Retrieval most critical (-12.76 F1 when removed)
      \item[\tiny-] Evolution essential (-11.76 F1)
      \item[\tiny-] Multi-agent architecture improves (-5.95 F1)
    \end{itemize}}
\end{itemize}

\vspace{0.15cm}
\textbf{\scriptsize Conclusion}
\begin{itemize}
  \item[\tiny$\checkmark$] {\tiny MulVul addresses heterogeneity via specialized agents with cross-model prompt evolution}
  \item[\tiny$\checkmark$] {\tiny Retrieval augmentation mitigates hallucination and enables few-shot detection}
  \item[\tiny$\checkmark$] {\tiny Demonstrates scalability \& effectiveness for real-world vulnerability detection}
\end{itemize}

\vspace{0.1cm}
{\tiny \textbf{Limitations}: Single language (C/C++), computational cost (1+k LLM calls), potential misuse}
\end{column}
