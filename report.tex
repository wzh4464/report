\section{总体概述}
\begin{frame}
  \frametitle{情境图谱：AI时代的"元数据2.0"}

  \begin{block}{核心定义}
    情境图谱是对传统知识图谱的演进，将\textbf{多模态的非结构化决策过程}作为记忆，用图结构组织
  \end{block}

  \begin{block}{与RAG/知识图谱的关键区别}
    \begin{itemize}
      \item \textbf{传统知识图谱}：结构化三元组 \texttt{(实体, 关系, 实体)}，仅捕获"是什么"
      \item \textbf{传统RAG}：向量检索，关注语义相似度，缺乏逻辑关联
      \item \textbf{情境图谱}：捕获决策轨迹、上下文背景、时间动态和多模态信息，回答"为什么"
    \end{itemize}
  \end{block}

  \begin{block}{关键特征}
    \begin{itemize}
      \item 记录非结构化决策过程（会议讨论、例外逻辑、审批链）
      \item 整合多模态数据（文本、图表、表格、时间序列）
      \item 图结构组织，支持多跳遍历和关系推理
      \item 为AI提供可解释的决策上下文
    \end{itemize}
  \end{block}
\end{frame}

\section{Foundation Capital：捕获决策轨迹的万亿美元机遇}
\begin{frame}
  \frametitle{Foundation Capital：万亿美元机遇 \cite{foundationcapital2024context}}

  \begin{block}{核心理念}
    传统记录系统（CRM、ERP）仅捕获最终数据状态（"是什么"），\textbf{情境图谱捕获决策轨迹}（"为什么"）
  \end{block}

  \begin{block}{决策轨迹包含什么？}
    \begin{itemize}
      \item Slack讨论、会议记录中的\textbf{例外逻辑}
      \item 过去案例的\textbf{先例和审批背景}
      \item 跨系统的综合信息（Salesforce + Zendesk + Slack）
      \item 系统外的\textbf{审批链}和决策理由
    \end{itemize}
  \end{block}

  \begin{block}{设计逻辑与商业价值}
    \begin{itemize}
      \item 坐落在\textbf{执行路径}中，实时记录输入数据、策略、异常路径和审批人
      \item 积累成为\textbf{可查询的先例库}
      \item 让AI能够参考过去案例处理复杂边界情况
    \end{itemize}
  \end{block}
\end{frame}

\section{CGR³：增强知识表示与推理范式}
\begin{frame}
  \frametitle{CGR³：上下文图谱推理范式 \cite{xu2024cgr3}}

  \begin{block}{核心定义}
    对传统三元组知识图谱的扩展，融入\textbf{时间动态、地理位置和来源信息}，提供更精准的知识理解
  \end{block}

  \begin{block}{CGR³范式：三阶段推理框架}
    \begin{enumerate}
      \item \textbf{检索 (Retrieve)}：候选实体和上下文
      \item \textbf{排序 (Rank)}：候选实体
      \item \textbf{推理 (Reason)}：判断信息是否充足
    \end{enumerate}
  \end{block}

  \begin{block}{LLM的作用}
    \begin{itemize}
      \item LLM擅长处理情境图谱提供的\textbf{非结构化背景信息}
      \item 显著提升知识图谱补全（KGC）和问答（KGQA）的准确性
      \item 能够理解和推理复杂的上下文关系
    \end{itemize}
  \end{block}
\end{frame}

\section{Atlan：主动基础设施的元数据2.0}
\begin{frame}
  \frametitle{Atlan：元数据2.0 \cite{atlan2024context}}

  \begin{block}{定义}
    融合了\textbf{操作元数据、治理策略和决策追溯}的增强型知识图谱，揭示数据流动的完整"操作现实"
  \end{block}

  \begin{block}{关键组件}
    \begin{enumerate}
      \item 图数据库基础（RDF或属性图）
      \item 元数据捕获层（查询解析、日志跟踪、API集成）
      \item 语义富化层（技术元数据→业务上下文）
      \item 上下文激活层（为AI提供决策支持）
    \end{enumerate}
  \end{block}

  \begin{block}{与传统工具的差异}
    传统线性血缘工具仅追踪"数据来自哪里"，而情境图谱支持\textbf{多跳遍历}，回答"为什么做出这个决定"和"哪些政策在当时生效"
  \end{block}
\end{frame}

\section{FlexRule：对"决策考古学"的批判}
\begin{frame}
  \frametitle{FlexRule：批判性思考 \cite{flexrule2024critique}}

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{设计缺陷}
        单纯依赖历史决策轨迹：将\textbf{"先例误认为政策"}
      \end{alertblock}

      \begin{block}{"决策考古学"陷阱}
        \begin{itemize}
          \item 从历史归纳规则，非明确意图
          \item 无法区分有效与无效逻辑
          \item "先例黑箱"：只能说"统计相似"
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{block}{替代方案：决策中心法}
        \small
        轨迹是\textbf{"收据"}，非\textbf{"食谱"}

        \textbf{五层架构：}
        \begin{enumerate}
          \item \textbf{意图层}：DMN/DDG显式定义决策逻辑
          \item \textbf{流程层}：CDM管理决策触发时机
          \item \textbf{输入层}：LiveContext提供决策就绪数据
          \item \textbf{痕迹层}：事后捕获，仅用于审计和学习
          \item \textbf{学习层}：显式优化，避免决策漂移
        \end{enumerate}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\section{i-SCOOP：信息的"结缔组织"}
\begin{frame}
  \frametitle{i-SCOOP：连接信息的结缔组织 \cite{iscoop2024context}}

  \begin{block}{核心隐喻}
    情境图谱捕捉信息中的\textbf{"结缔组织"}，将零散的组织知识转化为AI可以导航的逻辑路径
  \end{block}

  \begin{block}{技术洞察}
    向LLM提供\textbf{Cypher或RDF}等结构化格式的语境，比简单的CSV效果更好
    \begin{itemize}
      \item \textbf{原因}：结构本身就承载了语义价值
      \item 图结构明确表达了实体间的关系和依赖
      \item 支持推理和逻辑导航
    \end{itemize}
  \end{block}

  \begin{block}{未来前沿：时间关系}
    \begin{itemize}
      \item \textbf{时间维度}是真理判定的下一前沿
      \item 通过捕捉时间关系，AI可以判断信息是新鲜的还是陈旧的
      \item 区分经过反复证实的知识 vs. 孤立的说法
    \end{itemize}
  \end{block}
\end{frame}

\section{TrustGraph：上下文工程与互操作性}
\begin{frame}
  \frametitle{TrustGraph：标准与实现视角 \cite{trustgraph2024manifesto}}

  \begin{block}{定义}
    一种为AI使用而优化的三元组数据表示
  \end{block}

  \begin{block}{技术选择：RDF vs. 属性图}
    \begin{itemize}
      \item \textbf{RDF栈}（RDF, RDFS, OWL）
      \begin{itemize}
        \item 复杂但功能强大
        \item 支持形式化推理和互操作性
        \item 适合监管环境和复杂本体
      \end{itemize}
      \item \textbf{属性图}（Neo4j等）
      \begin{itemize}
        \item 更易理解和开发
        \item 查询性能优秀
        \item 适合应用开发者
      \end{itemize}
    \end{itemize}
  \end{block}

  \begin{block}{范式转移：LLM时代的语义网}
    LLM的出现使得我们可以处理\textbf{动态本体}，AI可以动态生成检索逻辑，实现语义网多年未竟的互操作性愿景
  \end{block}
\end{frame}

\section{Microsoft GraphRAG}
\begin{frame}
  \frametitle{Microsoft GraphRAG：工程化路径 \cite{microsoft2024graphrag}}

  \begin{block}{核心痛点：传统RAG的上下文碎片化}
    向量检索只看语义相似度，却漏掉了逻辑关联
  \end{block}

  \begin{block}{工程方法}
    \begin{enumerate}
      \item \textbf{实体提取与关系映射}：使用LLM分析文本单元
      \item \textbf{层级聚类}：利用\textbf{Leiden或Louvain算法}进行社区检测
      \item \textbf{全局感知摘要}：为每个社区生成摘要，实现宏观理解
    \end{enumerate}
  \end{block}

  \begin{block}{专用框架}
    \begin{itemize}
      \item \textbf{HopRAG}：逻辑感知检索，支持多跳推理
      \item \textbf{HippoRAG}：受大脑海马体启发的联想记忆
      \item \textbf{GNN-RAG}：神经子图处理，结合图神经网络
    \end{itemize}
  \end{block}
\end{frame}

\section{研究方向与实践应用}

\begin{frame}
  \frametitle{轨迹与过程取证 (Trajectory \& Process Forensics)}

  \begin{block}{核心理念}
    关注智能体\textbf{如何}到达解决方案，而不仅仅是\textbf{是否}到达。将智能体的执行路径（轨迹）作为主要研究对象
  \end{block}

  \begin{block}{三个关键研究方向}
    \begin{enumerate}
      \item \textbf{思维-行动一致性}：研究智能体的内部推理轨迹（CoT）是否与外部工具使用相符
      \begin{itemize}
        \item 常见失败：计划"检查文件大小"，实际执行 \texttt{rm -rf} 命令
      \end{itemize}

      \item \textbf{循环检测与停滞分析}：检测"语义循环"，智能体反复运行相同失败测试或编辑同一文件而无进展

      \item \textbf{分步效率（轨迹评分）}：为每一步分配成本，计算"最小可行轨迹"——修复错误的理论最短路径
    \end{enumerate}
  \end{block}
\end{frame}

\begin{frame}
  \frametitle{Context Graph 的定位}

  \begin{block}{Agent 的核心组成}
    Agent = LLM + 工具调用能力 + 循环执行
  \end{block}

  \begin{block}{为什么需要 Context Graph？}
    \begin{itemize}
      \item Agent 需要记住过去的经验 → 需要 \textbf{Memory}
      \item Memory = 存储 Agent 的历史信息，在需要时检索出来
      \item \textbf{Context Graph = 用图结构组织 Agent 的经验}
    \end{itemize}
  \end{block}

  \begin{block}{核心优势}
    将多模态的非结构化决策过程（会议讨论、工具调用、失败尝试）转化为结构化的图记忆，支持复杂推理和经验学习
  \end{block}
\end{frame}

\begin{frame}
  \frametitle{相关工作 (1/2)}

  \begin{block}{ExpeL (AAAI 2024) \cite{zhao2024expel}}
    \textbf{主要目的}：解决 LLM Agent 无法跨任务学习的问题
    \begin{itemize}
      \item 自主收集经验并使用自然语言提取知识
      \item 推理时调用提取的洞察和过去经验
    \end{itemize}
  \end{block}

  \begin{block}{A-Mem (NeurIPS 2025) \cite{xu2025amem}}
    \textbf{主要目的}：解决现有 Memory 系统的组织结构问题
    \begin{itemize}
      \item 传统 RAG 只是简单的存储-检索，缺乏组织
      \item 现有 graph 方案结构固定，适应性差
      \item 需要能\textbf{动态自组织}的记忆系统
      \item 采用 Zettelkasten 方法构建互联知识网络
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}
  \frametitle{相关工作 (2/2)}

  \begin{block}{可训练图记忆 (arXiv 2025) \cite{xia2025trainable}}
    \textbf{主要目的}：解决现有经验学习方法的两个核心问题
    \begin{enumerate}
      \item \textbf{隐式记忆}（训练）：灾难性遗忘 + 不可解释
      \item \textbf{显式记忆}（prompting）：缺乏适应性 + 无法评估策略好坏
    \end{enumerate}
    \textbf{目标}：构建可训练的图记忆，结合两者优点
    \begin{itemize}
      \item 多层图记忆框架：状态机 + 元认知策略
      \item 基于强化学习的权重优化
    \end{itemize}
  \end{block}

  \begin{block}{失败学习 (arXiv 2025) \cite{zhu2025fail}}
    研究 LLM Agent 在哪里失败以及如何从失败中学习
    \begin{itemize}
      \item 提出 AgentErrorTaxonomy 错误分类体系
      \item AgentErrorBench 失败轨迹数据集
      \item AgentDebug 调试框架，24\%准确率提升
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}
  \frametitle{Coding Agent 任务分类}

  \begin{block}{Context Graph 在编程任务中的应用}
    用 Context Graph 解决/优化 Coding Agent 的什么方面？
  \end{block}

  \begin{block}{主要任务类型}
    \begin{enumerate}
      \item \textbf{代码生成}：从自然语言描述生成代码
      \item \textbf{Bug 修复}：识别和修复代码中的错误
      \item \textbf{代码重构}：改进代码结构而不改变功能
      \item \textbf{漏洞检测}：发现代码中的安全漏洞
      \item \textbf{代码理解}：分析和解释现有代码
    \end{enumerate}
  \end{block}
\end{frame}

\begin{frame}
  \frametitle{研究方向 1：Bug 修复}

  \begin{block}{为什么选择 Bug 修复？}
    Context Graph 的独特优势：
    \begin{itemize}
      \item \textbf{历史失败经验}：记录过去的失败尝试和原因
      \item \textbf{调试轨迹}：追踪 Bug 修复过程中的思维链和工具调用
      \item \textbf{相似问题模式}：通过图结构快速检索类似 Bug 的解决方案
      \item \textbf{多步推理}：Bug 修复通常需要多轮迭代，Context Graph 支持复杂的多跳推理
    \end{itemize}
  \end{block}

  \begin{block}{如何使用 Context Graph}
    \begin{enumerate}
      \item 将每次修复尝试（成功和失败）存储为图节点
      \item 建立节点间的关系：因果关系、时间序列、相似性
      \item 从历史经验中学习最优修复策略
      \item 避免重复失败的路径
    \end{enumerate}
  \end{block}
\end{frame}

\begin{frame}
  \frametitle{研究方向 2：漏洞检测 (Vulnerability Detection)}

  \begin{block}{Context Graph 在漏洞检测中的应用}

      \item \textbf{漏洞模式库}：构建已知漏洞的图结构表示
      \item \textbf{代码依赖分析}：追踪数据流和控制流
      \item \textbf{跨文件推理}：检测跨多个文件的复杂漏洞
      \item \textbf{历史漏洞学习}：从过去的漏洞中提取模式
    \end{itemize}
  \end{block}

  \begin{block}{潜在研究问题}
    \begin{itemize}
      \item 如何有效表示代码中的安全模式？
      \item 如何在图中编码时序和数据流信息？
      \item 如何从历史漏洞中泛化到新的漏洞类型？
      \item 如何减少误报率？
    \end{itemize}
  \end{block}
\end{frame}
