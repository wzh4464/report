\begin{frame}
  \frametitle{Theoretical Foundation: Matrix Rank}
  \begin{block}{Key Insight}
    \textbf{Matrix rank does not increase when cutting into smaller matrices}
  \end{block}

  \begin{itemize}
    \item For a matrix $A \in \mathbb{R}^{M \times N}$ partitioned into blocks:
          \begin{itemize}
            \item Each block $B_{ij}$ represents a submatrix
            \item $\text{rank}(B_{ij}) \leq \text{rank}(A)$
          \end{itemize}

    \item Probabilistic model ensures co-cluster preservation: \cite{wu2024}
          \begin{equation}
            P(\omega_k) \leq \exp \{-2[\phi_m(s^{(k)})^2 + \psi_n(t^{(k)})^2]\}
          \end{equation}

    \item Success probability of identifying co-clusters:
          \begin{equation}
            P \geq 1 - \exp \{-2T_p[\phi_m(s^{(k)})^2 + \psi_n(t^{(k)})^2]\}
          \end{equation}
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Extending AHPM to Tensors}
  \textbf{Key Insights for Extension:}
  \begin{enumerate}
    \item \textbf{Partitioning Strategy}
          \begin{itemize}
            \item Extend partitioning to multi-dimensional space
            \item For tensor $\mathcal{T} \in \mathbb{R}^{I_1 \times I_2 \times \cdots \times I_N}$, partition along each dimension
          \end{itemize}

    \item \textbf{Probabilistic Model Adaptation}
          \begin{itemize}
            \item Extend probability formulation for tensor co-clusters
            \item Consider multi-dimensional block configurations
          \end{itemize}

    \item \textbf{Hierarchical Merging Extension}
          \begin{itemize}
            \item Develop similarity measures for tensor co-clusters
            \item Design merge strategies for multi-dimensional blocks
          \end{itemize}
  \end{enumerate}
\end{frame}

\begin{frame}
  \frametitle{Mathematical Formulation for Tensor Extension}
  \begin{block}{Tensor Partitioning}
    \footnotesize
    For a tensor $\mathcal{T} \in \mathbb{R}^{I_1 \times I_2 \times \cdots \times I_N}$, partition into blocks:
    \begin{equation}
      \mathcal{B}_{i_1,i_2,\ldots,i_N}, \quad 1 \leq i_j \leq m_j
    \end{equation}
  \end{block}

  \begin{block}{Probabilistic Model for Tensor Co-clusters}
    \footnotesize
    Extend matrix probability model:
    \begin{equation}
      P(\omega_k) \leq \exp \left\{-2\sum_{j=1}^{N} \phi_{m_j}(s_j^{(k)})^2 \right\}
    \end{equation}
    where $s_j^{(k)}$ is the minimum size of co-cluster $C_k$ in the $j$-th dimension.
  \end{block}

  \begin{block}{Success Probability}
    \footnotesize
    \begin{equation}
      P \geq 1 - \exp \left\{-2T_p\sum_{j=1}^{N} \phi_{m_j}(s_j^{(k)})^2 \right\}
    \end{equation}
  \end{block}
\end{frame}

\begin{frame}
  \frametitle{Proposed Algorithm for Tensor Co-Clustering}
  \begin{algorithm}[H]
    \caption{Tensor AHPM}
    \begin{algorithmic}[1]
      \Require Tensor $\mathcal{T} \in \mathbb{R}^{I_1 \times I_2 \times \cdots \times I_N}$, Block sizes $\{\phi_{i_j}\}$, Thresholds
      \Ensure Co-clustered result $\mathcal{C}$
      \State Initialize block set $\mathcal{B}$ based on dimensions
      \State Calculate minimum sizes $s_j^{(k)}$ for each co-cluster
      \For{each potential co-cluster $C_k$}
      \State Calculate $P(\omega_k)$ using extended model
      \If{$P(\omega_k) < P_{thresh}$}
      \State Partition tensor $\mathcal{T}$ into blocks $\mathcal{B}$
      \State Perform co-clustering on each block
      \State Merge co-clustered results hierarchically
      \EndIf
      \EndFor
      \State \Return Aggregated co-clustered result $\mathcal{C}$
    \end{algorithmic}
  \end{algorithm}
\end{frame}
