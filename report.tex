\section{科研工作}
\begin{frame}
  \frametitle{TIM：训练数据影响的时间动态度量}
  \begin{figure}[ht]
    \centering
    \includegraphics[width=\textwidth]{tim.png}
    \vspace{-0.7cm}
    \caption{TIM 方法概览}
  \end{figure}
  \vspace{-0.5cm}
  \footnotesize TIM 包含两个阶段：\\
  \textbf{1）参数影响度量}：利用递归估计与影响传播算子 $P_t = I - \eta_t H^{[t]}$ 跟踪参数偏移；\\
  \textbf{2）参数影响投影}：通过查询向量 $q$ 将参数变化映射到函数响应（如测试损失）。\\
  结合不同训练窗口的影响分析，TIM 支持细粒度的时间序列洞察。
\end{frame}

\begin{frame}
  \frametitle{LiveVal：利用自适应参考点的时间感知数据估值}
  \begin{figure}[ht]
    \centering
    \includegraphics[width=\textwidth]{liveval1.png}
    \caption{LiveVal 方法概览}
    \label{fig:liveval_overview}
  \end{figure}
  \footnotesize
  \textbf{图1}：以最终参数为静态参考点，比较样本梯度更新后与目标参数的距离变化，用归一化分值 \(v_i^t\) 判断该样本是否让模型更接近最优。\\
  \textbf{图2}：通过自适应窗口 \(\delta_t\) 选取近未来参数作为参考点，动态调整窗口大小以适应不同训练阶段，实现实时、低内存的数据价值评估。
\end{frame}

\subsection{LMEraser：基于自适应提示的大模型遗忘}
\begin{frame}
  \frametitle{LMEraser：基于自适应提示的大模型遗忘}
  \begin{figure}
    \centering
    \includegraphics[width=0.99\textwidth]{lmeraser.png}
    \caption{LMEraser 框架示意}
  \end{figure}
  \footnotesize LMEraser 包含四个阶段：\\
  \textbf{1）数据划分}：将训练数据拆分为公共与私有子集，并在公共数据上预训练骨干模型；\\
  \textbf{2）聚类}：基于多样性对私有数据进行聚类；\\
  \textbf{3）微调}：对每个聚类的提示与分类头进行定制化微调；\\
  \textbf{4）再微调}：对受影响聚类的提示进行快速再训练，实现高效遗忘。
\end{frame}
