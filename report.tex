\begin{frame}
\frametitle{\small Introduction \& Problem Statement}
\vspace{-0.15cm}
\begin{columns}[t]
\begin{column}{0.48\textwidth}
\textbf{\scriptsize Background}
\begin{itemize}\setlength{\itemsep}{0pt}
  \item[\scriptsize$\bullet$] {\scriptsize Code vulnerabilities threaten software security}
  \item[\scriptsize$\bullet$] {\scriptsize LLMs show promise for detection}
  \item[\scriptsize$\bullet$] {\scriptsize CWE: 130 types in 10 categories}
\end{itemize}

\vspace{0.1cm}
\textbf{\scriptsize Key Challenges}
\begin{enumerate}\setlength{\itemsep}{1pt}
  \item[\scriptsize 1.] {\scriptsize \textbf{Heterogeneity}: Single model struggles with diverse patterns}
  \item[\scriptsize 2.] {\scriptsize \textbf{Scalability}: Manual prompts for 100+ CWEs impractical}
  \item[\scriptsize 3.] {\scriptsize \textbf{Hallucination}: Multi-agent errors cascade}
  \item[\scriptsize 4.] {\scriptsize \textbf{Efficiency}: Querying all agents expensive}
\end{enumerate}
\end{column}

\begin{column}{0.48\textwidth}
\textbf{\scriptsize Proposed Solution: MulVul}
\begin{itemize}\setlength{\itemsep}{1pt}
  \item[\scriptsize$\star$] {\scriptsize \textbf{Router-Detector}: Coarse-to-fine strategy}
  \item[\scriptsize$\star$] {\scriptsize \textbf{Cross-Model Evolution}: Claude proposes, GPT-4o evaluates}
  \item[\scriptsize$\star$] {\scriptsize \textbf{Retrieval}: SCALE-based knowledge grounding}
\end{itemize}

\vspace{0.1cm}
\textbf{\scriptsize Main Contributions}
\begin{itemize}\setlength{\itemsep}{1pt}
  \item[\scriptsize$\diamond$] {\scriptsize Multi-agent framework with tool-augmented reasoning}
  \item[\scriptsize$\diamond$] {\scriptsize Automated prompt optimization}
  \item[\scriptsize$\diamond$] {\scriptsize SOTA: 34.79\% Macro-F1 (+41.5\% improvement)}
\end{itemize}
\end{column}
\end{columns}
\end{frame}

  \item[\scriptsize 2.] {\scriptsize \textbf{Prompt Evolution}:
    \begin{itemize}\setlength{\itemsep}{0pt}
      \item[\scriptsize-] Stage I: Optimize Router (Recall@k)
      \item[\scriptsize-] Stage II: Optimize Detectors (F1)
      \item[\scriptsize-] Generator mutates, Executor evaluates
    \end{itemize}}

\vspace{0.08cm}
\textbf{\scriptsize Online Detection}
\begin{itemize}\setlength{\itemsep}{1pt}
  \item[\scriptsize 1.] {\scriptsize \textbf{Router}: SCALE + Global retrieval $\to$ Top-k}
  \item[\scriptsize 2.] {\scriptsize \textbf{Detectors}: Contrastive retrieval $\to$ CWE types}
  \item[\scriptsize 3.] {\scriptsize \textbf{Isolation}: No inter-agent communication}
\end{itemize}
