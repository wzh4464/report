\section{Clustering vs Co-Clustering}

\begin{frame}
  \frametitle{Clustering: Overview and Limitations}
  \begin{columns}[T]
    \column{0.58\textwidth}
    \begin{block}{\footnotesize Overview}
      \footnotesize
      \begin{itemize}\itemsep0em
        \item Unsupervised learning technique for pattern recognition
        \item Objective: Maximize intra-cluster similarity and inter-cluster dissimilarity
        \item Key algorithms: K-means, Hierarchical, DBSCAN, Gaussian Mixture Models
      \end{itemize}
      % \footnotesize
      % Given $X = \{x_1, ..., x_n\}$, find $K$ clusters $C = \{C_1, ..., C_K\}$ such that:
      % \begin{enumerate}\itemsep0em
      %   \item $C_i \neq \emptyset$ for all $i = 1, ..., K$
      %   \item $\bigcup_{i=1}^K C_i = X$
      %   \item $C_i \cap C_j = \emptyset$ for $i \neq j$
      % \end{enumerate}
      % Objective: Minimize $\sum_{i=1}^K \sum_{x \in C_i} \text{distance}(x, \mu_i)$\\
      % Where $\mu_i$ is the centroid of cluster $C_i$
    \end{block}

    \begin{block}{\footnotesize Limitations}
      \footnotesize
      \begin{enumerate}\itemsep0em
        \footnotesize
        \item Global feature relevance assumption
              \begin{itemize}\itemsep0em
                \footnotesize
                % \item Problem: Assumes equal importance of all features
                \item Impact: Misses subspace clusters
                \item Example: Different word subsets in document clustering
              \end{itemize}
        \item Inability to capture local patterns
              \begin{itemize}\itemsep0em
                \footnotesize
                % \item Problem: Focus on global similarity measures
                \item Impact: Overlooks locally coherent patterns
                \item Example: Gene expression patterns in specific conditions
              \end{itemize}
      \end{enumerate}
    \end{block}

    \column{0.4\textwidth}
    \begin{figure}
      \includegraphics[width=\textwidth]{cluster.png}
      \captionof{figure}{\footnotesize Clustering illustration}
    \end{figure}
  \end{columns}
\end{frame}

\begin{frame}
  \frametitle{Co-Clustering}
  \begin{itemize}
    \item \textbf{Co-Clustering} groups both data points and features simultaneously.
    \item Captures hidden patterns across two dimensions (rows and columns).
    \item Suitable for complex, high-dimensional datasets.
  \end{itemize}
  \begin{block}{Special Advantage}
    Reveals deeper insights by analyzing data and feature interactions.
  \end{block}
  \begin{block}{Applications}
    \begin{itemize}
      \item Text mining: Clustering documents and words.
      \item Collaborative filtering: User and item clustering for recommendations.
      \item Bioinformatics: Grouping genes and conditions in gene expression data.
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}
  \frametitle{Co-Clustering Overview}
  \begin{columns}
    \column{0.55\textwidth}
    \begin{block}{\footnotesize Overview}
      \footnotesize
      \begin{itemize}
        \item Simultaneously clusters rows and columns of a matrix
        \item Captures relationships between data points and features
        \item Key algorithms: Spectral Co-clustering, NMF-based methods, Probabilistic models
      \end{itemize}
    \end{block}
    \begin{block}{\footnotesize Formal Definition}
      \footnotesize
      Given a matrix $X$ of size $m \times n$, find $K$ row clusters $R = \{R_1, ..., R_K\}$ and $L$ column clusters $C = \{C_1, ..., C_L\}$ such that:
      \begin{enumerate}
        \item $R_i \neq \emptyset$ for all $i = 1, ..., K$
        \item $C_j \neq \emptyset$ for all $j = 1, ..., L$
        \item $X_{ij} \in R_i \times C_j$
      \end{enumerate}
      Objective: Minimize $\sum_{i=1}^K \sum_{j=1}^L \text{similarity}(R_i, C_j)$
    \end{block}

\end{frame}
\section{Problem Statement}

\begin{frame}
  \frametitle{Challenges in Co-clustering Large-Scale Data}
  % \begin{block}{Key Challenges}
  \textbf{Key Challenges:}
  \begin{enumerate}
    \item \textbf{High Computational Complexity}
          \begin{itemize}\itemsep0em
            \item Simultaneous analysis of row and column relationships
            \item Exponential growth with data size
            \item Evaluates all possible row-column combinations
          \end{itemize}

    \item \textbf{Significant Communication Overhead}
          \begin{itemize}\itemsep0em
            \item Frequent exchange of intermediate results in partitioned data
            \item Iterative optimization requires continuous updates
            \item Communication becomes a major bottleneck
          \end{itemize}

    \item \textbf{Dependency on Sparse Matrices}
          \begin{itemize}\itemsep0em
            \item Traditional algorithms optimized for sparse matrices
            \item Real-world data often dense (most elements non-zero)
            \item Dense matrices lack computational shortcuts
          \end{itemize}
  \end{enumerate}
  % \end{block}
\end{frame}

\section{Methodology}

\begin{frame}
  \frametitle{Large Matrix Partitioning}
  \textbf{Addresses:} High Computational Complexity
  \begin{itemize}
    \item Probabilistic model for optimal partitioning
    \item Theorem: Lower bound on detection probability
          \[ P \ge 1 - \exp \left\{ -2 T_p [\phi m (s^{(k)})^2 + \psi n (t^{(k)})^2] \right\} \]
    \item Balances computational efficiency and co-cluster integrity
    \item Enables parallel processing of smaller submatrices
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Hierarchical Co-cluster Merging}
  \textbf{Addresses:} Significant Communication Overhead
  \begin{columns}
    \column{0.5\textwidth}
    \begin{itemize}
      \item Iterative merging process
      \item Starts with most similar clusters
      \item Adaptive threshold adjustment
      \item Reduces need for frequent intermediate result exchanges
      \item Ensures consistency and reduces redundancy
    \end{itemize}

    \column{0.5\textwidth}
    \begin{tikzpicture}[
        cluster/.style={circle, draw, minimum size=0.8cm},

    \end{tikzpicture}
  \end{columns}
\end{frame}

\begin{frame}
  \frametitle{Flexible Atom-co-clustering Framework}
  \textbf{Addresses:} Dependency on Sparse Matrices
  \begin{itemize}
    \item Compatible with various atom-co-clustering methods
    \item Example: Graph-based Spectral Co-clustering Algorithm (SCC)
          \begin{itemize}
            \item Constructs bipartite graph
            \item Uses graph Laplacian eigenvectors
            \item Applies k-means to singular vectors
          \end{itemize}
    \item Can be adapted for both sparse and dense matrices
  \end{itemize}
\end{frame}
