\subsection{Shape-Biased Ellipse Detection Network}
\begin{frame}
  \frametitle{Shape-Biased Ellipse Detection Network \cite{li2022ShapebiasedEllipseDetection}}
  \begin{columns}
    \column{0.6\textwidth}
    \begin{itemize}
      \item \textbf{Aims}: Improve ellipse detection in noisy, occluded, real-world images.
      \item \textbf{Method}: EllipseNet uses multitask learning for localization, contour detection, and parameter estimation, using 2D Gaussian models and Wasserstein loss.
      \item \textbf{Key Learnings}: Multitask learning improves accuracy and focus, especially through the auxiliary edge task, outperforming traditional methods.
    \end{itemize}
    \column{0.6\textwidth}
    \centering
    \includegraphics[width=\textwidth]{architecture.png}
  \end{columns}
\end{frame}

\begin{frame}
  \frametitle{Multitask Learning and Why Not End-to-End?}
  \begin{columns}
    \column{0.5\textwidth}
    \textbf{Multitask Learning (MTL)}:
    \begin{itemize}
      \item Shares features across tasks.
      \item \textbf{Tasks in EllipseNet}:
            \begin{itemize}
              \item Localization: Detects bounding boxes.
              \item Contour Detection: Focuses on ellipse edges.
              \item Parameter Prediction: Estimates ellipse parameters.
            \end{itemize}
      \item \textbf{Benefits}: Higher accuracy, better generalization, efficiency.
    \end{itemize}

    \column{0.5\textwidth}
    \textbf{Why Not End-to-End?}:
    \begin{itemize}
      \item \textbf{Specialization}: Tasks need different focuses.
      \item \textbf{Complexity}: Handles varying scales and occlusions better.
      \item \textbf{Feature Extraction}: MTL improves edge and shape focus.
      \item \textbf{Auxiliary Tasks}: Improves performance with shape-focused regularization.
    \end{itemize}
  \end{columns}
\end{frame}

\subsection{Advanced Ellipse Fitting Algorithm}
\begin{frame}
  \frametitle{Advanced Ellipse Fitting Algorithm \cite{zhang2021AdvancedEllipseFitting}}

  \begin{itemize}
    \item \textbf{Aims}: Improve accuracy and speed of ellipse fitting, particularly in computer vision and automatic manufacturing, using real-world noisy data.
    \item \textbf{Method}: A hybrid approach combining the BFGS algorithm and ADMM method to solve ellipse fitting as an optimization problem, reducing iterations and improving convergence.
    \item \textbf{Key Learnings}: This hybrid algorithm shows faster convergence and higher accuracy compared to traditional methods like Hough Transform and other regression-based techniques.
  \end{itemize}

\end{frame}

\subsection{A Method for Measuring the Inclination of Forgings}
\begin{frame}
  \frametitle{A Method for Measuring the Inclination of Forgings \cite{lu2023MethodMeasuringInclination}}

  \begin{itemize}
    \item \textbf{Aims}: Propose a method for measuring the inclination of forgings using an improved ellipse fitting algorithm.
    \item \textbf{Method}: The method uses a laser rangefinder to measure forging surfaces, random data point sampling to eliminate outliers, and an improved Particle Swarm Optimization (PSO) algorithm for fitting ellipses to calculate inclination angles.
    \item \textbf{Key Learnings}: The approach provides accurate and efficient ellipse fitting, with errors less than 0.05° compared to traditional methods, making it highly suitable for industrial applications.
  \end{itemize}

\end{frame}

\subsection{Morphometric Measurement of Fish Blood Cell}
\begin{frame}
  \frametitle{Morphometric Measurement of Fish Blood Cell \cite{ahmed2024MorphometricMeasurementFish}}

  \begin{itemize}
    \item \textbf{Aims}: Develop an automated method to measure morphometric parameters of fish blood cells using image processing and ellipse fitting techniques.
    \item \textbf{Method}: The method combines image segmentation and ellipse fitting to measure cell and nucleus areas in fish blood cells. The algorithm is evaluated against manual measurements and compared with existing methods.
    \item \textbf{Key Learnings}: The proposed method significantly reduces error compared to manual and traditional methods, providing more accurate and efficient cell area measurements, especially in complex scenarios with overlapping or broken cell borders.
  \end{itemize}

\end{frame}
