% 第1页：核心挑战与解决思路
\begin{frame}{核心挑战与解决思路}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
    \begin{block}{{\color{HuaweiRed}问题：人工编写Prompt的瓶颈}}
      \begin{itemize}
        \item \textbf{质量局限性:} 专家知识存在主观偏差，难以保证全局最优。
        \item \textbf{覆盖不完整:} 大量漏洞类型(CWE)缺乏高质量的专家模板。
        \item \textbf{知识难迁移:} 已有模板无法有效泛化到新的漏洞类型。
        \item \textbf{维护成本高:} 模板的更新和扩展严重依赖人工，效率低下。
      \end{itemize}
    \end{block}


    \column{0.5\textwidth}
    \begin{alertblock}{目标}
      实现从 \textbf{人工专家设计} 到 \textbf{算法自动化生成} 高质量Prompt模板的转变。
    \end{alertblock}
    \begin{block}{核心思路：进化算法 + MoP}
      \centering
      \begin{tikzpicture}[
          scale=0.6, transform shape, node distance=0.8cm and 1.2cm, auto,
          box/.style={rectangle, draw, rounded corners, fill=blue!15, align=center, font=\small, minimum width=2.0cm, minimum height=0.7cm},
          highlight/.style={rectangle, draw, rounded corners, fill=red!20, align=center, font=\small, minimum width=2.4cm, minimum height=0.8cm},
          note/.style={rectangle, draw, rounded corners, fill=green!15, align=center, font=\small, minimum width=1.8cm, minimum height=0.5cm},
          solidlink/.style={-latex, thin},
          dashedlink/.style={-latex, thin, dashed}
        ]

        % 两个起点
        \node[box] (expert) {专家种子\\Prompt};
        \node[box, right=1.8cm of expert] (general) {通用\\Prompt};

        % 引擎
        \node[highlight, below=1.0cm of $(expert)!0.5!(general)$] (engine) {\textbf{进化引擎}\\(GA/DE)};

        % MoP 模板库
        \node[box, below=0.5cm of engine, minimum width=2.8cm] (mop) {MoP 模板库};

        % 目标场景/任务

\begin{frame}{技术架构与核心优势}
  \begin{columns}[T,onlytextwidth]
    \column{0.55\textwidth}
    \begin{block}{自动化生成与优化流程}
      \centering
      \begin{tikzpicture}[scale=0.9, transform shape, node distance=0.5cm,
          process/.style={rectangle, draw, rounded corners, fill=blue!20, text width=2.6cm, align=center, font=\normalsize},
          highlight/.style={rectangle, draw, rounded corners, fill=red!20, text width=2.6cm, align=center, font=\normalsize},
          solidlink/.style={},
          dashedlink/.style={dashed}]
        \node[process] (seed) {1. 专家种子};
        \node[highlight, below=0.5cm of seed] (mutate) {2. 语义变异};
        \node[highlight, below=of mutate] (cross) {3. 交叉重组};
        \node[process, right=0.5cm of cross] (eval) {适应度评估\\F1-Score};
        \node[highlight, below=of cross] (select) {4. 择优选择};
        \node[process, below=of select] (output) {5. 输出MoP库};

        \draw[solidlink, ->] (seed) -- (mutate);
        \draw[solidlink, ->] (mutate) -- (cross);
        \draw[solidlink, ->] (cross) -- (select);
        \draw[solidlink, ->] (select) -- (output);
        \draw[dashedlink, ->] (select) -- (eval);
        \draw[dashedlink, ->] (eval) -- (select);
        \draw[solidlink, ->] (select) to[out=180,in=180,looseness=3] (mutate);
        \node[draw=red, dashed, inner sep=3pt, fit=(mutate) (cross) (select)] (loop) {};
        \node[above=0 of loop, color=red, font=\tiny] {\textbf{进化循环}};
      \end{tikzpicture}
    \end{block}

    \column{0.43\textwidth}
    \begin{block}{核心优势}
      \begin{itemize}
        \item \textbf{全局最优化:} 突破人工编写的局部最优限制，发掘更优模板。
        \item \textbf{自动化扩展:} 为新漏洞类型快速、低成本地生成高质量Prompt。
        \item \textbf{可持续演进:} 可基于新数据自动迭代，保持模板的先进性。
        \item \textbf{保证一致性:} 算法驱动生成，确保模板风格和质量的统一。
      \end{itemize}
    \end{block}
  \end{columns}
\end{frame}
