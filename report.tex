\section{Selected Journals}
\subsection{IEEE Transactions on Instrumentation and Measurement}
\begin{frame}
  \frametitle{IEEE Transactions on Instrumentation and Measurement (TIM)}
  \begin{itemize}
    \item Impact factor (IF): 5.6
    \item Review time: 3-6 months (relatively fast)
    \item Non CCF-listed
    \item 40 papers related to ellipse topics
    \item Estimated acceptance rate of 50\%
  \end{itemize}

  \textbf{Advantages}:
  \begin{itemize}
    \item High impact factor
    \item Short and predictable review time (3-6 months)
    \item Fast decision phase (1-2 weeks for decisions)
  \end{itemize}

  \textbf{Disadvantages}:
  \begin{itemize}
    \item Recent trends suggest some slowing down in review times
  \end{itemize}

  \textbf{Example Papers}:
  \begin{itemize}
    \item Shape-biased ellipse detection network with auxiliary task.
    \item Morphometric measurement of fish blood cell: an image processing and ellipse fitting technique.
  \end{itemize}
\end{frame}


\subsection{Computer Vision and Image Understanding}
\begin{frame}
  \frametitle{Computer Vision and Image Understanding (CVIU)}
  \begin{itemize}
    \item Impact factor (IF): 4.3

    \item CCF-B listed (high recognition in computer vision)
    \item 22 papers related to ellipse topics
  \end{itemize}

  \textbf{Advantages}:
  \begin{itemize}
    \item Prestigious in the field of computer vision and image processing
    \item High-quality and constructive review feedback
  \end{itemize}

  \textbf{Disadvantages}:
  \begin{itemize}
    \item Long and unpredictable review time (can take 1-2 years)
    \item Communication issues with editors reported in some cases
  \end{itemize}

  \textbf{Example Papers}:
  \begin{itemize}
    \item Effective elliptic fitting for iris normalization.
    \item Robust ellipse fitting using geometric constraints for biomedical imaging.
  \end{itemize}
\end{frame}
