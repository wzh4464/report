\section{EvoPrompt}
\begin{frame}
  \frametitle{EvoPrompt: \cite{guo2023evoprompt}}
  \begin{itemize}
    \item Treats prompts as genetic sequences in evolutionary algorithms
    \item Core evolutionary operations:
          \begin{itemize}
            \item Mutation: Random modifications to prompt components
            \item Crossover: Combining elements from successful prompts
            \item Selection: Choosing best-performing prompts as parents
          \end{itemize}
    \item Optimization process:
          \begin{itemize}
            \item Initialize population of prompt candidates
            \item Evaluate performance on target tasks
            \item Select top performers
            \item Generate new candidates through evolution
            \item Iterate until convergence
          \end{itemize}
    \item Key innovation: Automated prompt engineering through evolutionary search
    \item \textbf{GitHub:} \href{https://github.com/beeevita/EvoPrompt}{beeevita/EvoPrompt}
  \end{itemize}
\end{frame}

\section{TextGrad}
\begin{frame}
  \frametitle{TextGrad: \cite{yuksekgonul2024textgrad}}
  \begin{itemize}
    \item Core concept: Text-based gradient descent
          \begin{itemize}
            \item Treats text as differentiable variables
            \item Uses LLM feedback as text gradients
            \item Natural language replaces numerical gradients
          \end{itemize}
    \item Optimization process:
          \begin{itemize}
            \item Forward pass: LLM evaluates solution quality
            \item Backward pass: LLM generates improvement feedback
            \item Update: Solution refined based on feedback
            \item Iterate until convergence
          \end{itemize}
    \item Key innovation: End-to-end text optimization through LLM feedback
    \item \textbf{GitHub:} \href{https://github.com/zou-group/textgrad}{zou-group/textgrad}
  \end{itemize}
\end{frame}


\section{DSPy}
\begin{frame}[fragile]
  \frametitle{DSPy: \cite{khattab2023dspy}}
  \begin{itemize}
    \item Core concept: Declarative prompt programming
    \item Example implementation:
          \begin{lstlisting}[language=Python, 
      basicstyle=\scriptsize\fontfamily{SFMono-TLF}\selectfont, 
      breaklines=true, lineskip=-1pt, aboveskip=2pt, belowskip=2pt,
      keywordstyle=\color{blue}, commentstyle=\color{gray},
      stringstyle=\color{red}, literate={\#}{\#}1]

# Model Configuration
configure_language_model(model='gpt-4-mini')
# Task Definition
class Task:
    input: text
    output: sentiment (positive/negative)
# Module Selection
classifier = ChainOfThought(Task)
# Usage
result = classifier(text='This movie is great!')
output: {
    sentiment: 'positive',
    reasoning: 'expresses enthusiasm'
}
# Optimization
optimizer = BootstrapFewShot()
optimized_classifier = optimizer.compile(
    classifier,
    metric=accuracy_metric
)
    \end{lstlisting}
    \item Key innovation: Compiler-based prompt optimization
    \item \textbf{GitHub:} \href{https://github.com/stanfordnlp/dspy}{stanfordnlp/dspy}
  \end{itemize}
\end{frame}

\section{APOHF}
\begin{frame}[fragile]
  \frametitle{APOHF: \cite{lin2024prompt}}
  \begin{itemize}
    \item One-sentence summary: "Which answer do you prefer?" - A simple yet powerful approach to optimize prompts through human preference feedback
    \item Core concept: Human preference-based optimization
          \begin{itemize}
            \item No numeric scores needed
            \item Works with black-box LLMs
            \item Inspired by dueling bandits
          \end{itemize}
    \item Key features:
          \begin{itemize}
            \item Compare prompt pairs
            \item Collect human preferences
            \item Update iteratively
            \item Efficient optimization
          \end{itemize}
  \end{itemize}
\end{frame}

    \item Core concept: Dual optimization
          \begin{itemize}
            \item Instruction Optimization (IO)
            \item Exemplar Optimization (EO)
            \item Synergistic combination
          \end{itemize}
    \item Key findings:
          \begin{itemize}
            \item EO can outperform IO

            \item Best results from combined approach
          \end{itemize}
  \end{itemize}
\end{frame}
