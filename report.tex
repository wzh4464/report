\section{Paper Reading: SCALE}

\begin{frame}
  \frametitle{Paper Overview: SCALE}
  \textbf{Paper:} SCALE: Constructing Structured Natural Language Comment Trees for Software Vulnerability Detection \cite{wen2024scale}.
  
  \vspace{0.5em}
  \textbf{Problem:}
  \begin{itemize}
    \item Existing pre-trained models (e.g., CodeBERT) struggle to infer complex code logic (operators, pointers) and execution sequences.
  \end{itemize}
  
  \vspace{0.5em}
  \textbf{Solution (SCALE Framework):}
  \begin{itemize}
    \item \textbf{Comment Tree Construction:} Utilizes LLMs (e.g., ChatGPT) to generate summaries for code snippets, integrated into the AST.
    \item \textbf{Structured Natural Language Comment Tree (SCT):} Combines syntax templates with comments to explicitly capture execution flows.
    \item \textbf{Result:} Outperforms SOTA baselines on FFMPeg+Qemu, Reveal, and SVulD datasets.
  \end{itemize}
\end{frame}
