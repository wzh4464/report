\begin{frame}
  \frametitle{TIM: Time-Varying Influence Measurement of Training Data }
  \begin{figure}[ht]
    \centering
    \includegraphics[width=\textwidth]{tim.png}
    \vspace{-0.7cm}
    \caption{Overview of the TIM. }
  \end{figure}
  \vspace{-0.5cm}
  \footnotesize {TIM's two-stage approach: \\
    \textbf{1) Parameter Influence Measurement} uses recursive estimation with Influence Propagator $P_t = I - \eta_t H^{[t]}$ to track parameter deviations; \\
    \textbf{2) Parameter Influence Projection} maps parameter changes to functional responses (e.g., test loss) via query vectors $q$. \\
    By analyzing influence across different training windows, TIM enables fine-grained temporal influence analysis.}
\end{frame}

\begin{frame}
  \frametitle{LMEraser: Large Model Unlearning via Adaptive Prompt Tuning}
  % insert lmeraser.png
  \begin{figure}
    \centering
    \includegraphics[width=0.99\textwidth]{lmeraser.png}
    \caption{Overview of LMEraser.}
  \end{figure}
  \footnotesize { LMEraser's four-stage approach:   \\
    \textbf{1) Partitioning} training data into public and private datasets and pre- training backbone on public data.  \\
    \textbf{2) Clustering} private data based on diversity.\\
    \textbf{3) Tuning}  prompts and classifier heads for each cluster.\\
    \textbf{4) Re-Tuning}  prompt for affected cluster achieving efficient unlearning.\\
  }
\end{frame}
