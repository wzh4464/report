\begin{frame}{标准映射与落地参考}
  \begin{itemize}
    \item \textbf{MITRE CWE}：1099（命名不一致）、546（Suspicious Comment）等，可用于安全视角的缺陷归档
    \item \textbf{CERT Secure Coding}：DCL01-C/DCL04-C/EXP56-CPP 等覆盖变量、声明、语言链接
    \item \textbf{行业/社区规范}：Google \& Oracle Java Style、PEP 8、MISRA、ESLint/Prettier/clang-format 规则库
    \item \textbf{参考资料来源}：项目治理手册、静态分析规则库、标准化指南文档，支撑门禁规则配置
  \end{itemize}
\end{frame}

\section{类别 4：架构度量与 Code Smells}
\begin{frame}{覆盖率概览}
