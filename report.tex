\begin{frame}
  \frametitle{Key Progress This Week}
  \begin{itemize}
    \item \textbf{Generalization:} Achieved 80\% accuracy on unseen dev set.
    \item \textbf{Challenge:} Adding \textit{correct} caused high confusion with \textit{others}.
    \item \textbf{Solution:} Introduced a binary stage (\textit{vulnerable} vs. \textit{benign}), reaching \textbf{F1 = 0.92}.
    \item \textbf{Status:} 1\% scale experiment running; 10\% setting under test.
    \item \textbf{Prompt Control:} Growth issue mitigated by capping at 5,000 tokens, then summarizing while keeping task and format unchanged.
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Insights \& Next Steps}
  \begin{block}{Insights}
    \begin{itemize}
      \item Binary classification greatly reduces confusion, providing a cleaner pipeline.
      \item Summarization strategy prevents prompt explosion without hurting task fidelity.
    \end{itemize}
  \end{block}

  \begin{block}{Next Steps}
    \begin{itemize}
      \item Scale binary-first pipeline to full experiments.
      \item Monitor summarization impact on accuracy at higher percentages.
      \item Explore finer distinctions between \textit{correct} and \textit{others}.
    \end{itemize}
  \end{block}
\end{frame}
