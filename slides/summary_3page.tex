  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{行业演进路线}
        \scriptsize
        \textbf{第一代 (2020-2022): 向量 RAG}
        \begin{itemize}
            \item 向量检索 + 文档切片
            \item \textcolor{dotred}{痛点：无结构、无时序、无关系}
        \end{itemize}

        \vspace{0.05cm}
        \textbf{第二代 (2023): GraphRAG (Microsoft)}
        \begin{itemize}
            \item 静态知识图谱 + 社区摘要
            \item \textcolor{dotred}{痛点：批量构建、无法增量更新}
        \end{itemize}

        \vspace{0.05cm}
        \textbf{第三代 (2024-): Context Graph}
        \begin{itemize}
            \item \textcolor{darkgreen}{时序感知 + Schema-free + 增量更新}
            \item 专为 Agent Memory 设计
        \end{itemize}
      \end{block}

      \vspace{0.05cm}

      \begin{alertblock}{核心定位}
        \scriptsize
        Context Graph ≠ 通用知识图谱
        \newline 而是\textbf{为 LLM Agent 设计的记忆层架构}
        \newline 本质：存储"AI 应该记住什么"，而非"世界是什么样"
      \end{alertblock}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{block}{关键技术差异}
        \scriptsize
        \begin{tabular}{lll}
        \toprule
        \textbf{维度} & \textbf{传统 KG} & \textbf{Context Graph} \\
        \midrule
        Schema & 预定义 & \textcolor{darkgreen}{运行时自适应} \\
        更新 & 批量 ETL & \textcolor{darkgreen}{毫秒级增量} \\
        时间 & 单时间戳 & \textcolor{darkgreen}{双时序追溯} \\
        输出 & 结构化数据 & \textcolor{darkgreen}{Token-aware} \\
        \bottomrule
        \end{tabular}
      \end{block}

      \vspace{0.05cm}

      \begin{block}{性能数据 (Zep Benchmark)}

        \begin{itemize}

      \end{block}

      \vspace{0.05cm}
      \begin{tcolorbox}[colback=layer1]

    \end{column}
  \end{columns}

\begin{frame}
  \frametitle{2. 核心方案：Context Graph 可生长记忆架构}

  \begin{tcolorbox}[colback=layer1, title=\textbf{方案本质：为 LLM Agent 设计的工作记忆层}]
    \footnotesize
    借鉴 \textbf{Zep/Graphiti (记忆存储)} 与 \textbf{LangGraph (流程控制)}，构建 Coding 专用的 \textbf{Schema-free 时序图谱}。
  \end{tcolorbox}

  \vspace{0.1cm}

  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{block}{架构设计：显式化"经验路径"}
        \scriptsize
        \textbf{1. 节点 (Nodes) - 沉淀痕迹}
        \begin{itemize}
            \item \texttt{Episode}: 对话/操作流（时序有效期）
            \item \texttt{Entity}: 代码实体（运行时推断类型）
            \item \texttt{Insight}: \textbf{经验胶囊}（\textcolor{darkgreen}{主动抽象}规则，如"X错误 $\to$ Y解法"）
        \end{itemize}

        \vspace{0.05cm}

        \textbf{2. 有向边 (Directed Edges) - 锁定逻辑}
        \begin{itemize}
            \item \texttt{NEXT} (时序): 记录操作流，复用成功路径
            \item \texttt{CAUSES} (因果): 根因分析，多跳推理
            \item \texttt{DEPENDS\_ON}: 代码静态依赖（AST）
        \end{itemize}

        \vspace{0.05cm}
        \textbf{3. 双时序建模}：事实有效期 + 系统记录时间，支持"当时为什么这样决策"的追溯。
      \end{block}
    \end{column}

    \begin{column}{0.42\textwidth}
      \begin{block}{研发场景适配}
        \scriptsize
        \begin{tabular}{ll}
        \toprule
        \textbf{场景} & \textbf{Context Graph 解法} \\
        \midrule
        需求变更 & Schema-free 自适应 \\
        决策追溯 & 双时序可查历史 \\
        多人协作 & 跨 session 共享 \\
        代码+文档 & Episode $\to$ Entity 统一 \\
        \bottomrule
        \end{tabular}
      \end{block}

      \vspace{0.05cm}

      \begin{alertblock}{差异化竞争力}
        \scriptsize
        \textbf{Code KG (静态) + Context Graph (动态)}
        \begin{itemize}
            \item Code KG: "代码结构是什么"（AST、调用关系）
            \item Context Graph: "为什么这样写"（决策历史、需求演变）
        \end{itemize}
        同时理解\textcolor{darkgreen}{代码 + 背后决策}，形成壁垒。
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}
  \frametitle{3. 竞争格局与战略价值}

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{技术路线选择}
        \scriptsize
        \textbf{传统动态 KG 路线 \textcolor{dotred}{(不推荐)}}
        \begin{itemize}
            \item 需预定义本体，Schema 变更代价高
            \item 批量 ETL，无法适配快速迭代
            \item 定位模糊："又一个知识图谱应用"
        \end{itemize}

        \vspace{0.05cm}
        \textbf{Context Graph 路线 \textcolor{darkgreen}{(推荐)}}
        \begin{itemize}
            \item \textcolor{darkgreen}{技术代际领先}：对齐 2024-2025 行业范式
            \item Schema-free 降低建模成本
            \item 产品故事清晰："AI Agent 时代的记忆架构"
            \item 生态兼容：LangChain、LlamaIndex、MCP
        \end{itemize}
      \end{block}

      \vspace{0.05cm}

      \begin{alertblock}{竞品对标}
        \scriptsize
        \begin{itemize}
            \item 传统工具厂商（RAG）：我们有\textbf{关系推理 + 时序追溯}
            \item AI 创业公司（通用 Memory）：我们\textbf{深耕研发场景}
            \item 大厂（暴力长 context）：我们\textbf{更省 token、可私有部署}
        \end{itemize}
      \end{alertblock}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{block}{落地路线}
        \scriptsize
        \textbf{Phase 1 (MVP)}：基于 Graphiti 开源版快速验证
        \begin{itemize}
            \item 场景：SWE-bench Lite
            \item 目标：主动总结失败模式，不再重复犯错
        \end{itemize}

        \vspace{0.05cm}
        \textbf{Phase 2 (产品化)}：定制研发领域 Entity Types
        \begin{itemize}
            \item 场景：C/C++ 遗留系统
            \item 技术：AST 依赖 + Context Graph 协同
        \end{itemize}

        \vspace{0.05cm}
        \textbf{Phase 3 (差异化)}：多 Agent 协作 + 知识迁移
        \begin{itemize}
            \item 跨项目迁移成熟图谱
            \item 项目级知识沉淀与复用
        \end{itemize}
      \end{block}

      \vspace{0.05cm}

      \begin{tcolorbox}[colback=layer1, title=\textbf{业务价值}]

        \newline \textbf{增效}：Bug 解决率 +20-30\%
        \newline \textbf{资产化}：算力 $\to$ 企业私有知识资产
      \end{tcolorbox}
    \end{column}
  \end{columns}
