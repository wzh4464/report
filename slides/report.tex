  \end{block}

\section{Context Graph 概念框架}
%==============================================================================

\begin{frame}
  \frametitle{Context Graph：定义与定位}

  \begin{block}{核心定义 \cite{foundationcapital2024context}}
    Context Graph 是由\textbf{决策痕迹}（decision traces）编织成的活体记录结构，
    将实体、时间和先例连接成可查询的网络，使 AI Agent 能够理解"为什么这个决策被允许发生"。
  \end{block}

  \vspace{0.3cm}

  \begin{block}{与传统 KG 的区别}
    \small
    \begin{tabular}{p{2.2cm}|p{4cm}|p{4.5cm}}
      \textbf{维度} & \textbf{传统 KG} & \textbf{Context Graph} \\
      \hline
      设计目标 & 人类专家查询 & AI 模型消费 \\
      核心要素 & 实体 + 关系 & 实体 + 关系 + \textbf{决策事件} \\
      时间处理 & 无或单时态 & 双时态（有效时间+事务时间） \\
      溯源机制 & 可选 & \textbf{核心组件} \\
      更新方式 & 批量导入 & 实时/增量更新 \\
    \end{tabular}
  \end{block}

  \begin{block}{三个视角}
    \small
    \textbf{学术视角}：GraphRAG 算法创新 \quad
    \textbf{产业视角}：决策痕迹捕获 \quad
    \textbf{Agent 视角}：结构化经验记忆
  \end{block}
\end{frame}

\begin{frame}
  \frametitle{GraphRAG vs Context Graph}

  \begin{alertblock}{核心区分}
    \textbf{GraphRAG 是检索技术，Context Graph 是架构愿景}。两者共享"图"结构，但解决不同层次的问题
  \end{alertblock}

  \vspace{0.1cm}

  \footnotesize
  \renewcommand{\arraystretch}{1.3}
  \centering
  \begin{tabular}{>{\centering}p{2cm}|>{\centering}p{4.3cm}|>{\centering\arraybackslash}p{4.8cm}}
    \textbf{维度} & \textbf{GraphRAG} & \textbf{Context Graph} \\
    \hline
    定位 & 检索技术 & 架构愿景 \\
    \hline
    目标 & 改进 LLM 检索质量 & 捕获决策上下文 \\
    \hline
    核心数据 & 静态文档语料 & 跨系统实时状态 + 决策事件 \\
    \hline
    时态处理 & 静态快照 & 双时态（有效时间+事务时间） \\
    \hline
    溯源能力 & 引用来源文档 & 追溯完整决策推理路径 \\
    \hline
    成熟度 & \textcolor{darkgreen}{2024-2025 学术验证充分} & \textcolor{red}{需要架构层面重新设计} \\
  \end{tabular}

  \vspace{0.2cm}

  \normalsize
  \begin{block}{类比}
    GraphRAG 之于 Context Graph $\approx$ TCP/IP 之于互联网：\textbf{关键技术组件}，完整实现需要更多系统工程
  \end{block}
\end{frame}

\begin{frame}
  \frametitle{Context Graph 节点与边类型}

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{三种节点类型}
        \scriptsize
        \textbf{类型 1: Action Node（动作节点）}
        \begin{itemize}
          \item \texttt{action\_type}: read\_file | write\_file | run\_test
          \item \texttt{parameters}: \{file: "auth.py", ...\}
          \item \texttt{result}: success | failure | partial
          \item \texttt{cost}: token 消耗
        \end{itemize}

        \vspace{0.2cm}

        \textbf{类型 2: State Node（状态节点）}
        \begin{itemize}
          \item \texttt{description}: "发现 bug 在 line 42"
          \item \texttt{context}: 相关文件、变量
          \item \texttt{is\_terminal}: 是否终止状态
        \end{itemize}

        \vspace{0.2cm}

        \textbf{类型 3: Insight Node（洞察节点）}
        \begin{itemize}
          \item \texttt{insight}: "Django migration 问题需先检查 models"
          \item \texttt{confidence}: 0.85
          \item \texttt{source\_tasks}: [task\_1, task\_2, ...]
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{block}{四种边类型}
        \scriptsize
        \textbf{类型 1: NEXT（时序/因果）}
        \begin{itemize}
          \item Action\_A $\xrightarrow{\text{NEXT}}$ Action\_B
          \item 属性: success\_rate, avg\_reward, count
        \end{itemize}

        \vspace{0.2cm}

        \textbf{类型 2: CAUSES（因果）}
        \begin{itemize}
          \item Action\_X $\xrightarrow{\text{CAUSES}}$ Failure\_Y
          \item "修改了 config 导致测试失败"
        \end{itemize}

        \vspace{0.2cm}

        \textbf{类型 3: SIMILAR\_TO（相似）}
        \begin{itemize}
          \item Task\_1 $\xrightarrow{\text{SIMILAR\_TO}}$ Task\_2
          \item 用于跨任务经验迁移和模式复用
        \end{itemize}

        \vspace{0.2cm}

        \textbf{类型 4: ABSTRACTS（抽象）}
        \begin{itemize}
          \item [Action\_1, Action\_2, ...] $\rightarrow$ Insight
          \item 多个具体经验归纳为通用规则
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}
  \frametitle{Agent 演进史}

  \footnotesize
  \renewcommand{\arraystretch}{1.3}
  \centering
  \begin{tabular}{>{\centering}p{2cm}|p{3.5cm}|p{2.5cm}|p{3cm}}
    \textbf{系统} & \textbf{核心创新} & \textbf{记忆形式} & \textbf{局限} \\
    \hline
    \textbf{ReAct} (2023) & 思考+行动交替 & 无持久记忆 & 每次从零开始 \\
    \hline
    \textbf{Reflexion} \cite{shinn2023reflexion} & 语言化自我反思 & 反思列表（滑动窗口） & 经验无法跨任务 \\
    \hline
    \textbf{LATS} \cite{zhou2024lats} & MCTS 树搜索+价值函数 & 搜索树节点 & 树不持久化 \\
    \hline
    \textbf{Voyager} \cite{wang2024voyager} & 自动课程+技能库 & 代码技能库 & 扁平列表，无关联 \\
    \hline
    \textbf{Gen. Agents} \cite{park2023generative} & 记忆流+反思+计划 & Memory Stream & 线性增长，无图结构 \\
  \end{tabular}

  \vspace{0.3cm}

  \begin{alertblock}{演进趋势：Agent 越强，越需要结构化记忆}
    ReAct（无记忆）$\rightarrow$ Reflexion（反思列表）$\rightarrow$ LATS（搜索树）$\rightarrow$ Voyager（扁平技能库）$\rightarrow$ \textbf{Context Graph}（结构化知识图）
  \end{alertblock}
\end{frame}

\begin{frame}
  \frametitle{Agent 演进：Reflexion \& LATS}

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Reflexion \cite{shinn2023reflexion}}
        \small
        \textbf{语言化自我反思}（NeurIPS 2023）
        \begin{itemize}
          \item Actor 执行 $\rightarrow$ Evaluator 评估 $\rightarrow$ Self-Reflection 反思
          \item 反思存入 \texttt{mem}（长期记忆）
          \item AlfWorld +22\%, HumanEval +11\%
        \end{itemize}
        \vspace{0.1cm}
        \textcolor{red}{局限}：滑动窗口（1--3条），\\反思之间\textbf{无关联}，无法跨任务
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{block}{LATS \cite{zhou2024lats}}
        \small
        \textbf{MCTS 统一推理行动规划}（ICML 2024）
        \begin{itemize}
          \item 每步采样 $n$ 个候选动作
          \item UCT 平衡探索与利用
          \item LLM 自评估 + Self-Reflection
          \item HumanEval \textbf{92.7\%} (GPT-4 SOTA)
        \end{itemize}
        \vspace{0.1cm}
        \textcolor{red}{局限}：搜索树\textbf{不持久化}，\\每次任务从零重建
      \end{block}
    \end{column}
  \end{columns}

  \vspace{0.2cm}

  \begin{alertblock}{Context Graph = 持久化的搜索树 + 跨任务经验迁移}
    LATS 的树结构（节点=状态，边=动作，权重=价值）本质上就是 Context Graph 的雏形
  \end{alertblock}
\end{frame}

\begin{frame}
  \frametitle{Agent 演进：Voyager \& Generative Agents}

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Voyager \cite{wang2024voyager}}
        \small
        \textbf{开放世界终身学习 Agent}
        \begin{itemize}
          \item 自动课程生成（由易到难）
          \item GPT-4 生成可执行代码
          \item 技能存为代码+描述，语义检索复用
          \item 独特物品 3.3$\times$、里程碑速度 15.3$\times$
        \end{itemize}
        \vspace{0.1cm}
        \textcolor{red}{局限}：技能库是\textbf{扁平列表}，\\靠描述匹配检索，\\无法表达技能间的依赖关系
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{block}{Generative Agents \cite{park2023generative}}
        \small
        \textbf{可信人类行为模拟}（UIST 2023）
        \begin{itemize}
          \item \textbf{Memory Stream}：完整经历记录
          \item \textbf{Retrieval}：recency + importance + relevance
          \item \textbf{Reflection}：高层推断生成
          \item \textbf{Planning}：递归分解日程
        \end{itemize}
        \vspace{0.1cm}
        \textcolor{red}{局限}：Memory Stream \textbf{线性增长}，\\反思与观察混在同一列表
      \end{block}
    \end{column}
  \end{columns}

  \vspace{0.2cm}

  \begin{alertblock}{Agent $\rightarrow$ Context Graph 的必然}
    Voyager 的技能库需要\textbf{图结构}表达依赖；Gen. Agents 的 Memory Stream 需要\textbf{知识图谱}组织实体关系。
  \end{alertblock}

\begin{frame}

  \begin{columns}[T]
    \begin{column}{0.55\textwidth}

      \end{block}

          \item 新记忆触发旧记忆\textbf{属性更新}
          \item 自主决定存储、链接或遗忘
        \end{enumerate}
      \end{block}

      \begin{block}{实验结果（NeurIPS 2025）}
        \small
        复杂推理性能\textbf{翻倍}，Token 消耗降低 85--93\%，6 个模型均优于 SOTA
      \end{block}

      \begin{alertblock}{启示}
        \small
        \textbf{Agentic Memory}：记忆系统本身具有"代理性"，主动组织、链接、更新，而非被动存储-检索
      \end{alertblock}
    \end{column}

          \item \textcolor{red}{待完善}：因果关系溯源

      \end{block}

        \begin{itemize}
          \item Agent 路线的\textbf{动态交互}能力
          \item Memory 路线的\textbf{图结构}组织
          \item 领域特定的\textbf{Coding 知识}
        \end{itemize}
        \vspace{0.1cm}
        \textbf{核心创新}：代码静态逻辑（AST/Call Graph）\\
        $\oplus$ 动态决策轨迹（Context Graph）\\
        \textbf{=} 可生长的 Coding 记忆架构
      \end{alertblock}

      \begin{block}{FlexRule 警示 \cite{flexrule2024critique}}
        轨迹是``收据''，不是``食谱''\\
        $\Rightarrow$ CG 需区分\textbf{描述性} vs \textbf{规范性}知识
      \end{block}
    \end{column}

      \begin{block}{Context Graph 的作用}
        \begin{enumerate}
          \item 记录每次修复的完整轨迹（成功+失败）
          \item 自动提取 Bug 模式 $\rightarrow$ 修复策略规则
          \item 建立 Bug 模式 $\rightarrow$ 修复策略的映射
          \item 新任务时检索相似模式，避免重复犯错
        \end{enumerate}
      \end{block}

      \begin{block}{相关研究}
        \cite{zhu2025fail} 提出 AgentErrorTaxonomy，\\
        系统分类 Agent 的失败模式（推理错误、工具误用、循环等）
      \end{block}
    \end{column}

    \begin{column}{0.42\textwidth}
      \begin{alertblock}{关键研究问题}
        \begin{itemize}
          \item 如何表示"失败原因"？
          \item 如何判断两个 Bug 是否相似？
          \item 如何避免过时策略误导 Agent？
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}
  \frametitle{总结与下一步}

  \begin{block}{核心论点}
    \begin{itemize}
      \item \textbf{Agent 演进}（Reflexion $\rightarrow$ LATS $\rightarrow$ Voyager）：越来越需要结构化记忆
      \item \textbf{Memory 演进}（RAG $\rightarrow$ GraphRAG $\rightarrow$ A-Mem $\rightarrow$ Zep）：越来越趋向图结构
      \item 两条路的终点是\textbf{同一个}：\textbf{Context Graph} = 图结构 + 决策痕迹 + 动态学习
    \end{itemize}
  \end{block}

  \begin{block}{研究机会：Context Graph for Coding Agent}
    \begin{enumerate}
      \item \textbf{失败学习图}：将 Reflexion 的反思 + LATS 的搜索树 $\rightarrow$ 持久化经验图
      \item \textbf{技能依赖图}：将 Voyager 的技能库 $\rightarrow$ 有依赖关系的技能图谱
      \item \textbf{代码知识图}：融合 Zep 的时序 KG + 代码结构 $\rightarrow$ Coding Context Graph
    \end{enumerate}
  \end{block}

      \item 基线对比：CG vs Reflexion vs ExpeL 在跨任务学习上的差异
    \end{itemize}
